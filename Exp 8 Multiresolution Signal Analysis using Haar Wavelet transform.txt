Exp 8: Multiresolution Signal Analysis using Haar Wavelet transform


clc;
clear all;
close all;
N = 256;
n = 0:N-1;
x = sin(2*pi*50*n/N) + 0.5*sin(2*pi*20*n/N);

% Perform 3-level Haar wavelet decomposition
[C,L] = wavedec(x,3,'haar');

% Extract approximation and detail coefficients
A3 = appcoef(C,L,'haar',3);
D1 = detcoef(C,L,1);
D2 = detcoef(C,L,2);
D3 = detcoef(C,L,3);

% Plot Results
figure;
subplot(5,1,1);
plot(n,x);
title('Original Signal');
grid on;
subplot(5,1,2);
plot(A3);
title('Approximation Coefficients A3');
grid on;
subplot(5,1,3);
plot(D3);
title('Detail Coefficients D3');
grid on;
subplot(5,1,4);
plot(D2);
title('Detail Coefficients D2');
grid on;
subplot(5,1,5);
plot(D1);
title('Detail Coefficients D1');
grid on;

% Reconstruct signal
xr = waverec(C,L,'haar');
figure;
plot(n,x,'b',n,xr,'r--');
title('Original and Reconstructed Signal');
legend('Original','Reconstructed');
grid on;
