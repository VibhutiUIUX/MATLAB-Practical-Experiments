Exp 6: Adaptive noise Cancellation using LMS algorithm 


clc;
clear;
close all;

% Signal parameters
fs = 1000;
t = 0:1/fs:1-1/fs;
N = length(t);

% Original clean signal
s = sin(2*pi*50*t);

% Reference noise
x = randn(1,N);

% Noise correlated with reference signal
v = filter([1 0.5], 1, x);
v = 0.5 * v / std(v);

% Noisy input signal
d = s + v;

% LMS filter parameters
M = 16;
Mu = 0.01;
w = zeros(M,1);

% Initialize signals
y = zeros(1,N); % Estimated noise
e = zeros(1,N); % Noise-cancelled output

% LMS algorithm
for n = M:N
    xn = x(n:-1:n-M+1).';
    
    % Estimated noise
    y(n) = w.' * xn;
    
    % Error signal (noise-cancelled output)
    e(n) = d(n) - y(n);
    
    % LMS coefficient update
    w = w + Mu * e(n) * xn;
end

% Mean squared error
MSE = mean(e(M:N).^2);
fprintf('LMS filter order = %d\n', M);
fprintf('Step size = %.3f\n', Mu);
fprintf('Mean Square Error MSE = %.6f\n', MSE);

% Plot results
figure;

subplot(3,1,1)
plot(t,s,'LineWidth',1.2)
grid on
title('Original Signal')
xlabel('Time (s)')
ylabel('Amplitude')

subplot(3,1,2)
plot(t,d,'LineWidth',1.2)
grid on
title('Noisy Signal')
xlabel('Time (s)')
ylabel('Amplitude')

subplot(3,1,3)
plot(t,e,'LineWidth',1.2)
grid on
title('Noise free Signal')
xlabel('Time (s)')
ylabel('Amplitude')
