Exp 5: Implementation of forward linear prediction and estimation of prediction error


Code1: 

clc;
clear all;
close all;
Fs=1000;
t=0:1/Fs:0.05;
%Input Signal
x=sin(2*pi*100*t);
p=10; %Prediction Order
%Find Linear Prediction Coefficient
[a,PE] = lpc(x,p);
% Predicted Signal
x_pred=filter([0 -a(2:end)],1,x);
%Prediction Error
e=x-x_pred;
MSE=mean(e.^2);
% Display Results
fprintf('Prediction order=%d\n',p);
fprintf('Prediction error=%6f\n',PE);
fprintf('Mean Square error=%6f\n',MSE);
%Plot Results
figure;
subplot(3,1,1);
plot(t,x); %Original Signal
subplot(3,1,2);
plot(t,x_pred); %Predicted Signal
subplot(3,1,3);
plot(t,e); %Prediction Error
