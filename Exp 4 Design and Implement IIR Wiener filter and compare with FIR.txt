Exp 4: Design and Implement IIR Wiener filter and compare it with FIR


close; clc; clear;

%% Signal generation
fs=1000;
N=1000;
t= (0:N-1)/fs;

f1=50;
f2=120;

%% Discrete time signal
d = sin(2*pi*f1*t) + 0.5*sin(2*pi*f2*t);

%Gaussian noise
rng(1);
noise = 0.5*randn(1,N);

x = d + noise;

[Pxx,f] = pwelch(x,[],[],[],fs);

[Pdx,N] = cpsd(d,x,[],[],[],fs);

H = Pdx ./ Pxx;
omega = 2*pi*f/fs;

Nz = 4;
Np = 4;

[b_IIR, a_IIR] = invfreqz(H,omega, Nz, Np);

Y_IIR = filter(b_IIR, a_IIR,x);

MSE = mean((d-Y_IIR).^2);

fprintf("MSE = %.6f \n", MSE);

figure;
subplot(3,1,1);
plot(t,d);
title("Original Signal");
xlabel("time");
ylabel("amplitude");

subplot(3,1,2);
plot (t,x);
title("Noisy added Discrete Time Signal");
xlabel("time");
ylabel("amplitude");

subplot(3,1,3);
plot(t,Y_IIR);
title("IIR Filtered Signal");
xlabel("time");
ylabel("amplitude");

figure;
plot(t,d,LineWidth=1.5);

hold on;
plot(t,Y_IIR, LineWidth=1.5);
title("Comparison of Signal");
xlabel("time");
ylabel("amplitude");
legend("Original", "Filtered");
